newSwitch("Reset Desync", function(state)

	if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
		local hrp = plr.Character.HumanoidRootPart
		local pos = hrp.CFrame

		-- Teleport far away + snap back + anchor + velocity spam = full desync
		hrp.CFrame = pos + Vector3.new(0, 100000, 0)
		task.wait(0.05)
		hrp.CFrame = pos
		hrp.Anchored = true
		task.wait(0.12)
		hrp.Anchored = false

		for i = 1, 20 do
			hrp.Velocity = Vector3.new(math.random(-15000,15000), math.random(-15000,15000), math.random(-15000,15000))
			task.wait()
		end
		hrp.Velocity = Vector3.new(0,0,0)

		print("HAYDOS HUB - Reset Desync Activated | You are now fully desynced")
	end
end)

-- Player ESP
local function addESP(p)
	if p == plr or ESPTable[p] then return end
	local box = Instance.new("BoxHandleAdornment")
	box.Name = "HaydosESP"
	box.Size = Vector3.new(4,6,4)
	box.Color3 = Color3.fromRGB(0,255,255)
	box.Transparency = 0.4
	box.AlwaysOnTop = true
	box.ZIndex = 10
	box.Adornee = nil
	box.Parent = sg
	ESPTable[p] = box

	RunService.Heartbeat:Connect(function()
		if not p.Parent then return end
		local root = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
		local hum = p.Character and p.Character:FindFirstChild("Humanoid")
		if root and hum and hum.Health > 0 then
			box.Adornee = root
			box.Visible = ESPEnabled
		else
			box.Adornee = nil
		end
	end)

	p.CharacterAdded:Connect(function()
		task.wait(1)
		if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
			box.Adornee = p.Character.HumanoidRootPart
		end
	end)
end

for _, p in Players:GetPlayers() do if p ~= plr then addESP(p) end end
Players.PlayerAdded:Connect(addESP)
Players.PlayerRemoving:Connect(function(p)
	if ESPTable[p] then ESPTable[p]:Destroy() ESPTable[p] = nil end
end)

newSwitch("Player ESP", function(state)
	ESPEnabled = state
end)

local dragging, dragInput, dragStart, startPos

local function update(input)
	local delta = input.Position - dragStart
	if dragging == "main" then main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	elseif dragging == "tab" then tab.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y) end
end

top.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = "main" dragStart = i.Position startPos = main.Position end end)
tab.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = "tab" dragStart = i.Position startPos = tab.Position end end)
UIS.InputChanged:Connect(function(i) if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then update(i) end end)
UIS.InputEnded:Connect(function() dragging = false end)

local tween = TweenInfo.new(0.6, Enum.EasingStyle.Quint)
local openPos = UDim2.new(0.5, -220, 0, 40)
local hidePos = UDim2.new(0.5, -220, -1, 0)
local opened = true

local function show() TS:Create(main, tween, {Position = openPos}):Play() task.wait(0.1) tab.Visible = false opened = true end
local function hide() TS:Create(main, tween, {Position = hidePos}):Play() task.wait(0.5) tab.Visible = true opened = false end

minBtn.MouseButton1Click:Connect(function() if opened then hide() else show() end end)

UIS.InputBegan:Connect(function(i,gp)
	if not gp and i.KeyCode == Enum.KeyCode.Insert then
		if opened then hide() else show() end
	end
end)
